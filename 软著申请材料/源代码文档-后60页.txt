 461        SELECT

   1  const jwt = require('jsonwebtoken');
   2  
   3  /**
   4   * JWT 认证中间件
   5   * 从 Authorization header 中解析 token，验证并提取 userId
   6   */
   7  const authMiddleware = (req, res, next) => {
   8    try {
   9      // 从 header 中获取 token
  10      const authHeader = req.headers.authorization;
  11      
  12      if (!authHeader || !authHeader.startsWith('Bearer ')) {
  13        return res.status(401).json({
  14          success: false,
  15          message: '未提供认证令牌'
  16        });
  17      }
  18      
  19      // 提取 token
  20      const token = authHeader.replace('Bearer ', '');
  21      
  22      // 验证 token
  23      const decoded = jwt.verify(token, process.env.JWT_SECRET || 'your-secret-key');
  24      
  25      // 将 userId 添加到 request 对象
  26      req.userId = decoded.userId;
  27      req.openid = decoded.openid;
  28      
  29      next();
  30    } catch (error) {
  31      console.error('Token 验证失败:', error.message);
  32      
  33      if (error.name === 'TokenExpiredError') {
  34        return res.status(401).json({
  35          success: false,
  36          message: '认证令牌已过期，请重新登录'
  37        });
  38      }
  39      
  40      return res.status(401).json({
  41        success: false,
  42        message: '无效的认证令牌'
  43      });
  44    }
  45  };
  46  
  47  /**
  48   * 可选的 JWT 认证中间件
  49   * 如果有 token 则验证并提取用户信息，没有 token 则继续执行（允许匿名访问）
  50   */
  51  const optionalAuthMiddleware = (req, res, next) => {
  52    try {
  53      const authHeader = req.headers.authorization;
  54      
  55      // 没有 token，允许继续（匿名访问）
  56      if (!authHeader || !authHeader.startsWith('Bearer ')) {
  57        req.userId = null;
  58        req.openid = null;
  59        return next();
  60      }
  61      
  62      // 有 token，验证并提取用户信息
  63      const token = authHeader.replace('Bearer ', '');
  64      const decoded = jwt.verify(token, process.env.JWT_SECRET || 'your-secret-key');
  65      req.userId = decoded.userId;
  66      req.openid = decoded.openid;
  67      
  68      next();
  69    } catch (error) {
  70      console.error('Token 验证失败:', error.message);
  71      
  72      // Token 无效，但允许继续（降级为匿名访问）
  73      req.userId = null;
  74      req.openid = null;
  75      next();
  76    }
  77  };

  17     */
  18    async getAccessToken() {
  19      try {
  20        // 如果 token 还有效，直接返回
  21        if (this.accessToken && Date.now() < this.tokenExpireTime) {
  22          return this.accessToken;
  23        }
  24  
  25        // 获取新的 access_token
  26        const url = `https://api.weixin.qq.com/cgi-bin/token?grant_type=client_credential&appid=${this.appId}&secret=${this.appSecret}`;
  27        const response = await axios.get(url);
  28  
  29        if (response.data.access_token) {
  30          this.accessToken = response.data.access_token;
  31          // 提前 5 分钟过期，避免边界情况
  32          this.tokenExpireTime = Date.now() + (response.data.expires_in - 300) * 1000;
  33          console.log('✅ 获取微信 access_token 成功');
  34          return this.accessToken;
  35        } else {
  36          throw new Error(`获取 access_token 失败: ${JSON.stringify(response.data)}`);
  37        }
  38      } catch (error) {
  39        console.error('❌ 获取微信 access_token 失败:', error.message);
  40        throw error;
  41      }

 112  /**
 113   * 检查时间是否匹配（当前时间在目标时间之后的 10 分钟内）
 114   */
 115  function isTimeMatch(currentTime, targetTime) {
 116    const [currentHour, currentMinute] = currentTime.split(':').map(v => parseInt(v));
 117    const [targetHour, targetMinute] = targetTime.split(':').map(v => parseInt(v));
 118    
 119    const currentTotalMinutes = currentHour * 60 + currentMinute;
 120    const targetTotalMinutes = targetHour * 60 + targetMinute;
 121    
 122    // 计算时间差（当前时间 - 目标时间）
 123    const diff = currentTotalMinutes - targetTotalMinutes;
 124    
 125    // 如果当前时间在目标时间之后的 0-10 分钟内，则匹配
 126    return diff >= 0 && diff <= 10;
 127  }

  71  <script setup>
  72  import { ref, reactive, onMounted } from 'vue'
  73  import { ElMessage, ElMessageBox } from 'element-plus'
  74  import request from '@/utils/request'
  75  
  76  const loading = ref(false)
  77  const billList = ref([])
  78  
  79  const searchForm = reactive({
  80    userId: '',
  81    category: ''
  82  })
  83  
  84  const pagination = reactive({
  85    page: 1,
  86    pageSize: 10,
  87    total: 0
  88  })
  89  
  90  const fetchBills = async () => {
  91    try {
  92      loading.value = true
  93      const res = await request.get('/admin/bills', {
  94        params: {
  95          page: pagination.page,
  96          pageSize: pagination.pageSize,
  97          ...searchForm
  98        }
  99      })
 100      
 101      if (res.success) {
 102        billList.value = res.data.list
 103        pagination.total = res.data.total
 104      }
 105    } catch (error) {
 106      console.error('获取账单列表失败:', error)
 107    } finally {
 108      loading.value = false
 109    }
 110  }
 111  
 112  const handleSearch = () => {
 113    pagination.page = 1
 114    fetchBills()
 115  }
 116  
 117  const handleReset = () => {
 118    searchForm.userId = ''
 119    searchForm.category = ''
 120    handleSearch()
 121  }
 122  
 123  const handleDelete = (row) => {
 124    ElMessageBox.confirm('确定要删除这条账单吗？', '提示', {
 125      confirmButtonText: '确定',
 126      cancelButtonText: '取消',
 127      type: 'warning'
 128    }).then(async () => {
 129      try {
 130        const res = await request.delete(`/admin/bills/${row.id}`)
 131        if (res.success) {
 132          ElMessage.success('删除成功')
 133          fetchBills()
 134        }
 135      } catch (error) {
 136        console.error('删除失败:', error)
 137      }
 138    })
 139  }
 140  
 141  const formatDate = (date) => {
 142    if (!date) return '-'
 143    return new Date(date).toLocaleString('zh-CN')
 144  }
 145  
 146  onMounted(() => {
 147    fetchBills()
 148  })
